cmake_minimum_required(VERSION 3.20)
project(replay_queue LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED PATHS "${PYBIND_DIR}" NO_DEFAULT_PATH)
pybind11_add_module(_replay_queue bindings.cpp replay_queue.cpp)
target_compile_features(_replay_queue PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(_replay_queue PRIVATE /fp:strict)
else()
  target_compile_options(_replay_queue PRIVATE -ffp-contract=off)
endif()
install(TARGETS _replay_queue LIBRARY DESTINATION cloblab)
